\section*{Thin Disk Hydrostatic Equilibrium Derivation}

Assume
\begin{equation}
\varepsilon=\frac{H}{L}\ll1.
\end{equation}
For a thin disk,
\begin{equation}
V_z \ll V_r,V_\theta.
\end{equation}
\begin{equation}
B_z \ll B_r,B_\theta.
\end{equation}
\subsection*{Radial Momentum Balance}

\begin{equation}
\rho_0
\left[
\frac{\partial V_r}{\partial t}
+(\mathbf{V}\cdot\nabla)V_r
\right]
=
-\frac{\partial P_T}{\partial r}
-\frac{GM\rho_0}{r^2}
-\rho_0\frac{\partial\phi_{\rm sg}}{\partial r}
+\frac{(\mathbf{B}\cdot\nabla)B_r}{\mu_0}
+\rho_0\Omega^2 r.
\end{equation}

\subsection*{Vertical Momentum Balance}

\begin{equation}
\rho_0
\left[
\frac{\partial V_z}{\partial t}
+(\mathbf{V}\cdot\nabla)V_z
\right]
=
-\frac{\partial P}{\partial z}
-\frac{GM\rho_0 z}{r^3}
-\rho_0\frac{\partial\phi_{\rm sg}}{\partial z}
+\frac{(\mathbf{B}\cdot\nabla)B_z}{\mu_0}.
\end{equation}

Since $\varepsilon\ll1$, the vertical acceleration terms are negligible, giving
\begin{equation}
\frac{\partial P}{\partial z}
=
-\frac{GM\rho_0\, z}{r^3}.
\end{equation}

\subsection*{Vertical Pressure Profile}

Integrating over $z$,
\begin{equation}
P = -\frac{GM\rho_0}{2r^3}z^2 + C.
\end{equation}
Applying the free-surface boundary conditionat $z=H(r)$ $P=0$ ,
\begin{equation}
C = \frac{GM\rho_0 H^2(r)}{2r^3}.
\end{equation}
Hence
\begin{equation}
\boxed{
P = \frac{GM\rho_0}{2r^3}\left[H^2(r)-z^2\right].
}
\end{equation}

\subsection*{Depth-Averaged Pressure}

The disk occupies $z\in[0,H]$. By symmetry of $P$ in $z$, define
\begin{equation}
\bar{P} = \frac{1}{H}\int_0^{H} P\,dz.
\end{equation}
Substituting the pressure profile,
\begin{equation}
\bar{P}
= \frac{GM\rho_0}{2Hr^3}
\int_0^{H}\left[H^2(r)-z^2\right]dz.
\end{equation}
Evaluating the integral,
\begin{align}
\bar{P}
&= \frac{GM\rho_0}{2Hr^3}
\left[H^2 z - \frac{z^3}{3}\right]_0^{H} \\
&= \frac{GM\rho_0}{2Hr^3}
\left[H^3 - \frac{H^3}{3}\right] \\
&= \frac{GM\rho_0}{2Hr^3}\cdot\frac{2H^3}{3}.
\end{align}
Therefore,
\begin{equation}
\boxed{
\bar{P} = \frac{GM\rho_0 H^2(r)}{3r^3}.
}
\end{equation}

\subsection*{Depth-Averaged Radial Momentum Equation}

After vertical integration and using the thin-disk closure, the
depth-averaged radial momentum equation is:
\begin{equation}
\rho_0
\left[
\frac{\partial(H\bar{V}_r)}{\partial t}
+\left[(\mathbf{V}\cdot\nabla)V\,H\right]_r
-2\Omega V_\theta H
\right]
=
-\frac{\partial(H\bar{P})}{\partial r}+\rho_0\Omega^2 rH - \frac{\rho_0GMH}{r^2}
-\rho_0\left(\frac{\partial\phi}{\partial r}\right)_{\rm sg}
+\frac{\left[(\mathbf{B}\cdot\nabla)\mathbf{B}\,H\right]_r}{\mu_0}.
\end{equation}
Note that planetary gravity does not appear explicitly here
because it is already encoded in $\bar{P}$ through the
hydrostatic vertical balance.

\subsection*{Base State}

For the equilibrium state ${V}=0$, ${B}=0$.
Neglecting self-gravity,
\begin{equation}
\left(\frac{\partial\phi}{\partial r}\right)_{\rm sg}=0.
\end{equation}
The radial momentum equation reduces to:
\begin{equation}
0 = -\frac{\partial{P}}{\partial r} + \rho_0\Omega^2 r- \frac{\rho_0GM}{r^2}..
\end{equation}
Expanding the pressure gradient using the product rule,
\begin{equation}
\frac{\partial{P}}{\partial r}
= \rho_0\Omega^2 r\ - \frac{\rho_0GM}{r^2}.
\end{equation}


\subsection*{Keplerian Rotation}

For Keplerian rotation $\Omega^2 = GM/r^3$, the right-hand side
vanishes:
\begin{equation}
\frac{2H}{r^3}\frac{dH}{dr} - \frac{3H^2}{r^4} = 0.
\end{equation}
Dividing by $3H/r^3$:
\begin{equation}
\frac{dH}{dr} = \frac{H}{r}\cdot\frac{3}{2}.\\
\end{equation}
\begin{equation}
H=H_0\left(\frac{r}{r_0}\right)^\frac{3}{2}
\end{equation}

\section*{Derivation of the Linearised Matrix Equation}

\subsection*{1. Initial Linearised Equations}
Linearised equations about the background state $\vec{V}_0 = 0, \quad B_0 = 0, \quad H=H(r)$.

\textbf{Continuity Equation:}
\begin{equation}
\frac{\partial \eta}{\partial t} + \frac{1}{r} \frac{\partial}{\partial r} (H r V_r') + \frac{1}{r} \frac{\partial}{\partial \theta} (H V_\theta') = 0
\end{equation}

\textbf{$r$ Momentum (LMB) Equation:}
\begin{equation}
\frac{\partial}{\partial t}(H V_r') + GM \left[ \frac{2H}{r^3} \frac{\partial H}{\partial r} \eta + \frac{H^2}{r^3} \frac{\partial \eta}{\partial r} - \frac{3H^2\eta}{r^4} \right] - 2\Omega H V_\theta' - \frac{B_0}{\mu_0 \rho} B_r' + H \frac{\partial \phi'}{\partial r} = 0
\end{equation}

\textbf{$\theta$ Momentum (LMB) Equation:}
\begin{equation}
\frac{\partial}{\partial t} (H V_\theta') + \frac{GM H^2}{r^4} \frac{\partial \eta}{\partial \theta} + 2 \Omega V_r' H - \frac{B_0}{\mu_0 \rho} B_\theta' + \frac{H}{r} \frac{\partial \phi'}{\partial \theta} = 0
\end{equation}

\textbf{$r$ Induction Equation:}
\begin{equation}
\frac{\partial}{\partial t}(H B_r') + B_0 V_r' = 0
\end{equation}

\textbf{$\theta$ Induction Equation:}
\begin{equation}
\frac{\partial}{\partial t}(H B_\theta') + B_0 V_\theta' = 0
\end{equation}

\subsection*{2. Substitution of the Scale Height Profile}
Given $H = H_0 \left( \frac{r}{r_0} \right)^{3/2}$, the radial derivative is:
\begin{equation}
\frac{\partial H}{\partial r} = \frac{3H}{2r}
\end{equation}
Substituting this into the initial equations and dividing by $H$ yields the simplified system:

\textbf{Continuity:}
\begin{equation}
\frac{\partial \eta}{\partial t} + H \frac{\partial V_r'}{\partial r} + \frac{5H}{2r} V_r' + \frac{H}{r} \frac{\partial V_\theta'}{\partial \theta} = 0
\end{equation}

\textbf{$r$ Momentum:}
\begin{equation}
\frac{\partial V_r'}{\partial t} + GM \frac{H}{r^3} \frac{\partial \eta}{\partial r} - 2\Omega V_\theta' - \frac{B_0}{\mu_0 \rho H} B_r' + \frac{\partial \phi'}{\partial r} = 0
\end{equation}

\textbf{$\theta$ Momentum:}
\begin{equation}
\frac{\partial V_\theta'}{\partial t} + \frac{GM H}{r^4} \frac{\partial \eta}{\partial \theta} + 2 \Omega V_r' - \frac{B_0}{\mu_0 \rho H} B_\theta' + \frac{1}{r} \frac{\partial \phi'}{\partial \theta} = 0
\end{equation}

\textbf{Induction:}
\begin{align}
\frac{\partial B_r'}{\partial t} + \frac{B_0}{H} V_r' &= 0 \\
\frac{\partial B_\theta'}{\partial t} + \frac{B_0}{H} V_\theta' &= 0
\end{align}

\subsection*{3. Application of the Ansatz}
Applying the normal mode ansatz $q' = \tilde{q}\exp\left[i\left(\int k_r \, dr + m\theta - \omega t\right)\right]$ and the specified gravitational potential gradients:
\begin{align}
\frac{\partial\phi'}{\partial r} &\to -\frac{2\pi i G_N\rho\, k_r}{|k|}\tilde{\eta} \\
\frac{1}{r}\frac{\partial\phi'}{\partial\theta} &\to -\frac{2\pi i G_N\rho}{|k|}\frac{m}{r}\tilde{\eta}
\end{align}
As requested, we neglect the explicit $r$-dependence introduced by the scale height derivative ($i\frac{5k_r}{2r} \to 0$) in the dispersion relation. The equations transform into the algebraic system:

\begin{align}
-i\omega\tilde{\eta} + i H k_r \tilde{V}_r + \frac{imH}{r}\tilde{V}_\theta &= 0 \\
-i\omega\tilde{V}_r + \frac{i k_r GM H}{r^3}\tilde{\eta} - 2\Omega\tilde{V}_\theta - \frac{B_0}{\mu_0 \rho H}\tilde{B}_r - \frac{2\pi i G \rho k_r}{|k|}\tilde{\eta} &= 0 \\
-i\omega\tilde{V}_\theta + \frac{i m GM H}{r^4}\tilde{\eta} + 2\Omega\tilde{V}_r - \frac{B_0}{\mu_0 \rho H}\tilde{B}_\theta - \frac{2\pi i G \rho m}{|k| r}\tilde{\eta} &= 0 \\
-i\omega\tilde{B}_r + \frac{B_0}{H}\tilde{V}_r &= 0 \\
-i\omega\tilde{B}_\theta + \frac{B_0}{H}\tilde{V}_\theta &= 0
\end{align}

\subsection*{4. Elimination of Magnetic Field Perturbations}
Rearranging the induction equations provides the magnetic field variables in terms of velocity:
\begin{equation}
\tilde{B}_r = -i \frac{B_0}{\omega H} \tilde{V}_r, \quad \tilde{B}_\theta = -i \frac{B_0}{\omega H} \tilde{V}_\theta
\end{equation}
Substituting these into the momentum equations and grouping terms for $\tilde{V}_r, \tilde{V}_\theta,$ and $\tilde{\eta}$:

\textbf{$r$ Momentum:}
\begin{equation}
i k_r \left( \frac{GM H}{r^3} - \frac{2\pi G_N \rho}{|k|} \right) \tilde{\eta} - i \left( \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \right) \tilde{V}_r - 2\Omega\tilde{V}_\theta = 0
\end{equation}

\textbf{$\theta$ Momentum:}
\begin{equation}
\frac{im}{r} \left( \frac{GM H}{r^3} - \frac{2\pi G_N \rho}{|k|} \right) \tilde{\eta} + 2\Omega\tilde{V}_r - i \left( \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \right) \tilde{V}_\theta = 0
\end{equation}

\subsection*{5. Matrix Formulation}
Constructing the final system $M \mathbf{x} = 0$ for the primary variables $\mathbf{x} = [\tilde{\eta}, \tilde{V}_r, \tilde{V}_\theta]^T$ using the continuity and substituted momentum equations. To ensure proper fit on the page, the matrix is scaled.

\vspace{0.5cm}
\noindent
\resizebox{\textwidth}{!}{%
$
\begin{bmatrix}
-i\omega & i H k_r & \frac{imH}{r} \\[12pt]
i k_r \left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) & -i \left( \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \right) & -2\Omega \\[12pt]
\frac{im}{r} \left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) & 2\Omega & -i \left( \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \right)
\end{bmatrix}
\begin{bmatrix}
\tilde{\eta} \\[12pt]
\tilde{V}_r \\[12pt]
\tilde{V}_\theta
\end{bmatrix}
=
\begin{bmatrix}
0 \\[12pt]
0 \\[12pt]
0
\end{bmatrix}
$
}


\subsection*{6. Matrix Determinant Formulation}
To find the non-trivial solutions, compute $\det(M) = 0$. For compactness, define the modified frequency $\tilde{\omega}$ and the gravitational potential modifier $\Phi$:
\begin{align}
\tilde{\omega} &= \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \\
\Phi &= \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|}
\end{align}

Substituting these into the original matrix yields:
\begin{equation}
M = \begin{bmatrix}
-i\omega & i H k_r & \frac{imH}{r} \\[8pt]
i k_r \Phi & -i\tilde{\omega} & -2\Omega \\[8pt]
\frac{im}{r} \Phi & 2\Omega & -i\tilde{\omega}
\end{bmatrix}
\end{equation}

The determinant expands as:
\begin{equation}
\det(M) = M_{11}(M_{22}M_{33} - M_{23}M_{32}) - M_{12}(M_{21}M_{33} - M_{23}M_{31}) + M_{13}(M_{21}M_{32} - M_{22}M_{31}) = 0
\end{equation}

\subsection*{7. Exact Dispersion Relation} Let $k^2 = k_r^2 + \frac{m^2}{r^2}$:
\begin{equation}
\omega \left[ \left( \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \right)^2 - 4\Omega^2 \right] - H \left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) \left[ k^2 \left( \omega - \frac{B_0^2}{\mu_0 \rho \omega H^2} \right) + \frac{5m\Omega}{r^2} \right] = 0
\end{equation}

\subsection*{8. Polynomial Formulation}
Define the Alfvén frequency squared:
\begin{equation}
\omega_A^2 = \frac{B_0^2}{\mu_0 \rho H^2}
\end{equation}

The dispersion relation becomes:
\begin{equation}
\omega^3 - 2\omega_A^2\omega + \frac{\omega_A^4}{\omega} - 4\Omega^2\omega - H\Phi k^2 \omega + H\Phi k^2 \frac{\omega_A^2}{\omega} - H\Phi\frac{5m\Omega}{r^2} = 0
\end{equation}

Multiply the entire equation by $\omega$ to yield a quartic equation:
\begin{equation}
\omega^4 - \left[ 2\omega_A^2 + 4\Omega^2 + H\Phi k^2 \right] \omega^2 - \left( H\Phi\frac{5m\Omega}{r^2} \right) \omega + \omega_A^2 \left( \omega_A^2 + H\Phi k^2 \right) = 0
\end{equation}

The equation takes the standard polynomial form:
\begin{equation}
C_4\omega^4 + C_3\omega^3 + C_2\omega^2 + C_1\omega + C_0 = 0
\end{equation}

The expanded coefficients are:
\begin{align}
C_4 &= 1 \\
C_3 &= 0 \\
C_2 &= - \left[ 2\omega_A^2 + 4\Omega^2 + H \left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) k^2 \right] \\
C_1 &= - \frac{5m\Omega H}{r^2} \left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) \\
C_0 &= \omega_A^2 \left[ \omega_A^2 + H \left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) k^2 \right]
\end{align}

\section*{9. Deduction of Cases and the Rossby Mode}

\subsection*{Non-Magnetic Case ($B_0 = 0$)}
Setting $\omega_A = 0$, the $C_0$ term vanishes, and we can factor out $\omega$:
\begin{equation}
\omega \left[ \omega^3 - \left( 4\Omega^2 + H\Phi k^2 \right)\omega - H\Phi\frac{5m\Omega}{r^2} \right] = 0
\end{equation}

One root is trivially $\omega = 0$. The other roots satisfy the cubic:
\begin{equation}
\omega^3 - \left( 4\Omega^2 + H\Phi k^2 \right)\omega - H\Phi\frac{5m\Omega}{r^2} = 0
\end{equation}

If we consider the tightly wound limit ($k \gg m/r$), the high-frequency solutions are standard inertia-gravity (density) waves:
\begin{equation}
\omega^2 \approx 4\Omega^2 + H\Phi k^2 = 4\Omega^2 + H\left( \frac{GM H}{r^3} - \frac{2\pi G \rho}{|k|} \right) k^2
\end{equation}

\subsection*{The Rossby Mode Limit}
The Rossby mode is characterized by low frequencies ($\omega \ll \Omega$). In the non-magnetic limit ($B_0 = 0$), we drop the $\omega^3$ term in the cubic equation compared to the linear term:
\begin{equation}
-\left( 4\Omega^2 + H\Phi k^2 \right)\omega \approx H\Phi\frac{5m\Omega}{r^2}
\end{equation}
Yielding the standard Rossby mode frequency:
\begin{equation}
\omega_{\rm Rossby} \approx - \frac{H\Phi \frac{5m\Omega}{r^2}}{4\Omega^2 + H\Phi k^2}
\end{equation}
This clearly shows the requirement for azimuthal dependence ($m \neq 0$) to support the Rossby wave. In the purely radial limit ($m=0$), $\omega_{\rm Rossby} \to 0$.

\subsection*{Magnetic Case ($B_0 \neq 0$)}
With a non-zero magnetic field, $C_0 \neq 0$, and the true $\omega \to 0$ root is lifted. For low frequencies, we balance the lowest order terms in $\omega$ ($C_2 \omega^2 + C_1 \omega + C_0 \approx 0$):
\begin{equation}
- \left[ 2\omega_A^2 + 4\Omega^2 + H\Phi k^2 \right] \omega^2 - \left( H\Phi\frac{5m\Omega}{r^2} \right) \omega + \omega_A^2 \left( \omega_A^2 + H\Phi k^2 \right) \approx 0
\end{equation}
Assuming $4\Omega^2$ is dominant in the quadratic coefficient (as $\Omega \gg \omega_A$), we have:
\begin{equation}
4\Omega^2 \omega^2 + \left( H\Phi\frac{5m\Omega}{r^2} \right) \omega - \omega_A^2 \left( \omega_A^2 + H\Phi k^2 \right) \approx 0
\end{equation}
This quadratic equation provides two low-frequency roots, one of which represents a magnetically-modified Rossby mode, and the other is a new slow mode introduced by the magnetic tension. The presence of the $-C_0$ term can induce instabilities when $\omega_A^2 + H\Phi k^2 < 0$.
